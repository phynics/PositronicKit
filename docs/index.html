// Quick Start

// Package.swift
.package(url: "https://github.com/phynics/PositronicKit.git", from: "3.4.0")
import PositronicKit

let kit = PositronicKit(languageModel: myLLM)

// Choose the smallest operation tier that fits the feature:
let answer = try await kit.complete("Summarize this note.")  // tier 1: one-shot
let timeline = try await kit.timelineManager.createTimeline()
let driver = kit.openTimeline(timeline.id)               // tier 2: TimelineDriver
let agent = kit.agenticRuntime(                               // tier 4: agent loop
    timelineID: driver.timelineID,
    agentInstanceID: UUID()
)

// Tool Use

Hand a Tool conformance to the agent per turn — no workspace required:

let agent = kit.agenticRuntime(timelineId: driver.timelineID, agentInstanceId: UUID())
let stream = try await agent.run(message: "Greet Ada.", tools: [GreetTool()])

for try await event in stream {
    if case let .delta(.toolExecution(id, status)) = event { print("tool \(id): \(status)") }
    if let text = event.textContent { print(text, terminator: "") }
}

Or give the runtime a workspace root — every timeline gets its own filesystem workspace, and the built-in file tools (read, write, list) install automatically:

let kit = PositronicKit(configuration: .init(
    provider: .init(llmService: myLLM),
    persistence: .inMemory(),
    runtime: .init(workspaceRoot: workspaceRoot)  // filesystem tools installed per timeline
))

let timeline = try await kit.timelineManager.createTimeline(title: "Docs agent")

let stream = try await kit.run(.init(
    timelineId: timeline.id,
    message: "Read notes.md from the workspace and summarize it."
))

// Key Strengths

SwiftUI-Style Declarative Prompts

Author complex prompts as structured trees with a familiar body-composition DSL. Modifiers like .priority(...), .cachePolicy(...), and .compression(...) inherit automatically.

Smart Context Caching

PromptJournal tracks prompts dynamically, computing stable prefixes and volatile overlays to maximize LLM cache hit rates — lower latency, lower cost.

Zero-Latency Sidecar Directives

Fetch titles, sentiment, or summaries piggy-backed on the same single LLM request as the user-visible response. No extra round-trips.

Swift 6 Structured Concurrency

Fully thread-safe runtime built on actors and structured concurrency. Composable execution stages guarantee resource cleanup even on failure.

Completely Pluggable

Swap persistence engines, tool routers, and workspace resolvers freely. Anthropic, OpenAI, Ollama, and OpenRouter supported out of the box with zero runtime dependencies.

First-Class Linux Support

Compiles and tests seamlessly on Apple platforms and Linux via the bare Swift 6 toolchain, with an optimized Rust bridge for local MiniLM embeddings.
